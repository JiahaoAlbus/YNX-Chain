cmake_minimum_required(VERSION 3.18)
project(ynx_social_joint_profile_issuer LANGUAGES CXX)
# This module never imports a Go archive or changes the enclosing runtime.
# The original normal target supplies matching Auth headers, SAME-kernel Go
# frame/body definitions, and concrete registered Source/holder/custody ports.
if(NOT DEFINED YNX_SOCIAL_JOINT_AUTH_INCLUDE_DIR)
  message(FATAL_ERROR "Matching frozen original Auth include directory required")
endif()
if(NOT DEFINED YNX_SOCIAL_JOINT_ORIGINAL_SERVICES_TARGET OR
   NOT TARGET "${YNX_SOCIAL_JOINT_ORIGINAL_SERVICES_TARGET}")
  message(FATAL_ERROR "Actual registered-service adapter target required; no PUBLIC or unavailable fallback")
endif()
if(NOT DEFINED YNX_SOCIAL_JOINT_FINITE_CUSTODIAN_INCLUDE_DIR)
  message(FATAL_ERROR "Matching frozen finite custodian private include directory required")
endif()
if(NOT DEFINED YNX_SOCIAL_JOINT_MAIN_PAIR_INCLUDE_DIR)
  message(FATAL_ERROR "Matching frozen Main committed-pair private include directory required")
endif()
if(NOT DEFINED YNX_SOCIAL_JOINT_REGISTERED_MAIN_INCLUDE_DIR)
  message(FATAL_ERROR "Matching registered Main read private include directory required")
endif()
add_library(ynx_social_joint_profile_issuer OBJECT SocialJointIssuer.cpp SourceFrame.cpp CaptureLoan.cpp CapturedSource.cpp FiniteRole.cpp CommittedSocial.cpp MainPairRoute.cpp RegisteredMainPair.cpp)
target_compile_features(ynx_social_joint_profile_issuer PRIVATE cxx_std_17)
target_compile_options(ynx_social_joint_profile_issuer PRIVATE -Wall -Wextra -Werror)
target_include_directories(ynx_social_joint_profile_issuer PRIVATE "${YNX_SOCIAL_JOINT_AUTH_INCLUDE_DIR}" "${YNX_SOCIAL_JOINT_FINITE_CUSTODIAN_INCLUDE_DIR}" "${YNX_SOCIAL_JOINT_MAIN_PAIR_INCLUDE_DIR}" "${YNX_SOCIAL_JOINT_REGISTERED_MAIN_INCLUDE_DIR}")
target_link_libraries(ynx_social_joint_profile_issuer PRIVATE "${YNX_SOCIAL_JOINT_ORIGINAL_SERVICES_TARGET}")
if(DEFINED YNX_SOCIAL_JOINT_COMBINED_TARGET)
  if(NOT TARGET "${YNX_SOCIAL_JOINT_COMBINED_TARGET}")
    message(FATAL_ERROR "Original combined target is not present")
  endif()
  target_sources("${YNX_SOCIAL_JOINT_COMBINED_TARGET}" PRIVATE $<TARGET_OBJECTS:ynx_social_joint_profile_issuer>)
  target_link_libraries("${YNX_SOCIAL_JOINT_COMBINED_TARGET}" PRIVATE "${YNX_SOCIAL_JOINT_ORIGINAL_SERVICES_TARGET}")
endif()
