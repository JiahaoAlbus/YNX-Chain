cmake_minimum_required(VERSION 3.22)
project(ynx_original_ai_agent_v6 LANGUAGES CXX)
if(NOT DEFINED YNX_AI_AGENT_COMBINED_NATIVE_TARGET)
  message(FATAL_ERROR "A must select its one original combined native image")
endif()
if(NOT TARGET "${YNX_AI_AGENT_COMBINED_NATIVE_TARGET}")
  message(FATAL_ERROR "One existing combined target required; no second Go image")
endif()
if(NOT DEFINED YNX_AI_AGENT_MATCHED_AUTH_INCLUDE_ROOTS)
  message(FATAL_ERROR "Exact matched Auth capture/body/presence supplier required")
endif()
if(NOT DEFINED YNX_AI_AGENT_MATCHED_MATH_INCLUDE_ROOT)
  message(FATAL_ERROR "Exact sole original fixed35/E5 math headers required")
endif()
if(NOT DEFINED YNX_AI_AGENT_MATCHED_MATH_CPP)
  message(FATAL_ERROR "A must select the supplied sole fixed math CPP, not a fallback")
endif()
file(READ "${CMAKE_CURRENT_LIST_DIR}/INPUT_PINS.json" _ynx_agent_pins)
string(JSON _ynx_agent_math_sha GET "${_ynx_agent_pins}" actualFixedMathBody sha256)
file(SHA256 "${YNX_AI_AGENT_MATCHED_MATH_CPP}" _ynx_agent_selected_math_sha)
if(NOT _ynx_agent_math_sha STREQUAL _ynx_agent_selected_math_sha)
  message(FATAL_ERROR "Selected sole fixed AI math body pin differs")
endif()
list(GET YNX_AI_AGENT_MATCHED_AUTH_INCLUDE_ROOTS 0 _ynx_agent_auth_source_root)
string(JSON _ynx_agent_expected_source_root GET "${_ynx_agent_pins}" captureIncludeRoot)
if(NOT _ynx_agent_auth_source_root STREQUAL _ynx_agent_expected_source_root)
  message(FATAL_ERROR "Use the exact whole-matched Auth source root; no old header fallback")
endif()
set(_ynx_agent_expected_include_roots "${_ynx_agent_expected_source_root}")
string(JSON _ynx_agent_aux_count LENGTH "${_ynx_agent_pins}" normalAuxiliaryIncludeRoots)
math(EXPR _ynx_agent_aux_last "${_ynx_agent_aux_count} - 1")
foreach(_ynx_agent_i RANGE 0 ${_ynx_agent_aux_last})
  string(JSON _ynx_agent_aux GET "${_ynx_agent_pins}" normalAuxiliaryIncludeRoots ${_ynx_agent_i})
  list(APPEND _ynx_agent_expected_include_roots "${_ynx_agent_aux}")
endforeach()
if(NOT "${YNX_AI_AGENT_MATCHED_AUTH_INCLUDE_ROOTS}" STREQUAL "${_ynx_agent_expected_include_roots}")
  message(FATAL_ERROR "Auth include inventory must match the complete paired supplier")
endif()
foreach(_ynx_agent_artifact IN ITEMS AuthWhole AuthCaptureBody actualFixedMathH)
  string(JSON _ynx_agent_artifact_path GET "${_ynx_agent_pins}" ${_ynx_agent_artifact} path)
  string(JSON _ynx_agent_artifact_sha GET "${_ynx_agent_pins}" ${_ynx_agent_artifact} sha256)
  file(SHA256 "${_ynx_agent_artifact_path}" _ynx_agent_actual_sha)
  if(NOT _ynx_agent_actual_sha STREQUAL _ynx_agent_artifact_sha)
    message(FATAL_ERROR "Whole-matched Agent supplier changed: ${_ynx_agent_artifact}")
  endif()
endforeach()
foreach(_ynx_header IN ITEMS MobileAITaskRuntimePrivate.h MobileAISigningCapturePairedPrivate.hpp)
  if(_ynx_header STREQUAL "MobileAITaskRuntimePrivate.h")
    string(JSON _ynx_expected GET "${_ynx_agent_pins}" AuthRuntime sha256)
  else()
    string(JSON _ynx_expected GET "${_ynx_agent_pins}" pairedCaptureGetter sha256)
  endif()
  file(SHA256 "${_ynx_agent_auth_source_root}/${_ynx_header}" _ynx_actual)
  if(NOT _ynx_expected STREQUAL _ynx_actual)
    message(FATAL_ERROR "Matched Auth Agent header pin differs")
  endif()
endforeach()
add_library(ynx_original_ai_agent_v6 OBJECT AgentOperation.cpp RegisteredAgentResource.cpp "${YNX_AI_AGENT_MATCHED_MATH_CPP}")
set_target_properties(ynx_original_ai_agent_v6 PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(ynx_original_ai_agent_v6 PRIVATE cxx_std_20)
target_include_directories(ynx_original_ai_agent_v6 PRIVATE . .. ${YNX_AI_AGENT_MATCHED_AUTH_INCLUDE_ROOTS} ${YNX_AI_AGENT_MATCHED_MATH_INCLUDE_ROOT})
target_sources(${YNX_AI_AGENT_COMBINED_NATIVE_TARGET} PRIVATE $<TARGET_OBJECTS:ynx_original_ai_agent_v6>)
# Directory/SHA, Source/capture/approval, fixed AI math and neutral runtime strong
# bodies are selected once by original A. No archive/provider is imported here.
