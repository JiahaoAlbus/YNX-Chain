package com.ynx.social

import android.app.Activity
import android.os.Bundle
import android.widget.TextView
import org.json.JSONObject
import org.matrix.rustcomponents.sdk.MediaSource

// TEMPORARY ISOLATED SDK PROBE. Not product routing, login, media download or E2E.
class MediaSourceProbeActivity : Activity() {
  override fun onCreate(state: Bundle?) {
    super.onCreate(state)
    val report = try {
      val file = JSONObject().put("url", "mxc://example.org/public-fixture")
        .put("key", JSONObject().put("kty", "oct").put("key_ops", org.json.JSONArray(listOf("encrypt", "decrypt")))
          .put("alg", "A256CTR").put("k", "AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA").put("ext", true))
        .put("iv", "AAAAAAAAAAAAAAAAAAAAAA")
        .put("hashes", JSONObject().put("sha256", "AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA"))
        .put("v", "v2")
      val input = JSONObject().put("file", file).toString()
      val source = MediaSource.fromJson(input)
      try {
        val encoded = source.toJson()
        val parsed = JSONObject(encoded)
        val roundtrip = parsed.optJSONObject("file") ?: parsed
        check(roundtrip.getString("v") == "v2")
        check(roundtrip.getString("url") == source.url())
        check(roundtrip.getJSONObject("key").getString("k") == file.getJSONObject("key").getString("k"))
        check(roundtrip.getJSONObject("hashes").getString("sha256") == file.getJSONObject("hashes").getString("sha256"))
        check(roundtrip.getString("iv") == file.getString("iv"))
        check(com.ynx.social.matrix.MatrixMediaDescriptor.matches(file, roundtrip))
        val changed = JSONObject(file.toString()).put("iv", "AQAAAAAAAAAAAAAAAAAAAA")
        check(!com.ynx.social.matrix.MatrixMediaDescriptor.matches(changed, roundtrip))
        val duplicates = JSONObject(file.toString())
        duplicates.getJSONObject("key").put("key_ops", org.json.JSONArray(listOf("decrypt", "decrypt")))
        check(!com.ynx.social.matrix.MatrixMediaDescriptor.matches(duplicates, roundtrip))
        "PASS: real pinned SDK encrypted MediaSource roundtrip and owner descriptor checks.\nNo network/account/download/crypto-delivery.\n" + encoded
      } finally { source.close() }
    } catch (error: Throwable) {
      "FAIL: " + error.javaClass.name + ": " + error.message
    }
    setContentView(TextView(this).apply { text = report; textSize = 16f; setPadding(20, 32, 20, 20) })
  }
}
